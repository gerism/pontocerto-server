-- ============================================ -- PONTOCERTO — Schema inicial -- ============================================ -- Dados pessoais do atleta — permanentes, editáveis a qualquer momento. -- Um atleta só se cadastra uma vez aqui, e depois usa esse cadastro pra -- se inscrever em quantos eventos quiser. CREATE TABLE IF NOT EXISTS atletas ( id SERIAL PRIMARY KEY, device_id TEXT UNIQUE NOT NULL, -- identifica o aparelho, mesmo padrão dos outros apps nome TEXT NOT NULL, cpf TEXT UNIQUE NOT NULL, email TEXT NOT NULL, -- obrigatório pro Mercado Pago (payer_email na cobrança Pix) data_nascimento DATE NOT NULL, peso_kg NUMERIC(5,2), sexo TEXT CHECK (sexo IN ('M', 'F')), telefone TEXT NOT NULL, criado_em TIMESTAMP DEFAULT NOW(), atualizado_em TIMESTAMP DEFAULT NOW() ); -- Cada prova/corrida que você organiza. O código é o que o atleta digita -- no app pra encontrar e se inscrever no evento certo. CREATE TABLE IF NOT EXISTS eventos ( id SERIAL PRIMARY KEY, nome TEXT NOT NULL, codigo TEXT UNIQUE NOT NULL, -- ex: "CLD2026" — o que o atleta digita data_evento DATE NOT NULL, valor_inscricao NUMERIC(10,2) NOT NULL, oferece_camisa BOOLEAN DEFAULT FALSE, -- se esse evento dá camisa no kit ativo BOOLEAN DEFAULT TRUE, -- permite "fechar" um evento pra novas inscrições criado_em TIMESTAMP DEFAULT NOW() ); -- Categorias de idade definidas pelo organizador, por evento (cada -- evento pode ter faixas diferentes — não é fixo no sistema). CREATE TABLE IF NOT EXISTS categorias_evento ( id SERIAL PRIMARY KEY, evento_id INTEGER NOT NULL REFERENCES eventos(id) ON DELETE CASCADE, nome TEXT NOT NULL, -- ex: "15 a 20 anos" (o admin escolhe o texto) idade_min INTEGER NOT NULL, idade_max INTEGER NOT NULL, sexo TEXT CHECK (sexo IN ('M', 'F')) -- NULL = categoria mista (ambos os sexos) ); -- Liga um atleta a um evento específico. É aqui que fica o status do -- pagamento, a tag RFID vinculada (só existe/vale pra esse evento), e o -- resultado (largada/chegada/tempo) depois da corrida. CREATE TABLE IF NOT EXISTS inscricoes ( id SERIAL PRIMARY KEY, atleta_id INTEGER NOT NULL REFERENCES atletas(id) ON DELETE CASCADE, evento_id INTEGER NOT NULL REFERENCES eventos(id) ON DELETE CASCADE, categoria_id INTEGER REFERENCES categorias_evento(id), -- calculada automaticamente quando o pagamento é confirmado -- Pagamento (Mercado Pago Pix) pagamento_status TEXT NOT NULL DEFAULT 'pendente' CHECK (pagamento_status IN ('pendente', 'pago', 'cancelado')), mp_payment_id TEXT, -- id do pagamento no Mercado Pago, pra conferir no webhook -- Tag RFID vinculada a essa inscrição específica (não ao atleta em si, -- já que a tag pode ser reaproveitada/trocada entre eventos) tag_epc TEXT, -- Escolha de camisa do kit, quando o evento oferece (varia por evento, -- por isso fica na inscrição e não no cadastro permanente do atleta) quer_camisa BOOLEAN DEFAULT FALSE, camisa_tipo TEXT CHECK (camisa_tipo IN ('masculina', 'baby look')), camisa_tamanho TEXT CHECK (camisa_tamanho IN ('P', 'M', 'G')), -- Marcado pelo admin quando já entregou tag + camisa (kit completo) -- pro atleta — ajuda a saber quem já está pronto pra corrida. kit_entregue BOOLEAN DEFAULT FALSE, -- Resultado da corrida (preenchido pelo sistema de cronometragem) hora_largada TIMESTAMP, hora_chegada TIMESTAMP, tempo_total INTERVAL GENERATED ALWAYS AS (hora_chegada - hora_largada) STORED, criado_em TIMESTAMP DEFAULT NOW(), -- Um atleta só pode ter UMA inscrição por evento (evita pagar 2x sem querer) UNIQUE (atleta_id, evento_id) ); -- Índice pra buscar rápido "últimas 5 inscrições de um atleta" CREATE INDEX IF NOT EXISTS idx_inscricoes_atleta ON inscricoes (atleta_id, criado_em DESC); -- Índice pra achar rápido pela tag RFID durante a corrida (leitura em tempo real) CREATE INDEX IF NOT EXISTS idx_inscricoes_tag ON inscricoes (tag_epc);
